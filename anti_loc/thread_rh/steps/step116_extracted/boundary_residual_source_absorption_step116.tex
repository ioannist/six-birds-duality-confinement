\section{Residual sector}
\begin{definition}[Boundary-to-co-Poisson residual sector]
For a finite set of log-shift modes $\mathcal L_N$, define
\[
\mathcal R_N
=\overline{\operatorname{span}}\{\Ran |\Pi_{Y_a}\mathfrak B_{\ell,a}|:\ell\in\mathcal L_N\}\subseteq Y_a.
\]
Let $M_{R,N}:\mathbb C^{d_N}\to Y_a$ synthesize a finite dictionary for $\mathcal R_N$, and set
\[
G_{R,N}=M_{R,N}^*M_{R,N}.
\]
The finite residual form is
\[
\Xi^{\BC}_N\preceq \sigma_N G_{R,N}+T_N^{\BC},
\]
where $T_N^{\BC}$ is the finite-to-completed residual tail.
\end{definition}

\section{Residual coefficient visibility}
Let $D_N:\mathbb C^{I_N}\to H_N$ be a declared coefficient synthesis dictionary, typically Burnol/co-Poisson native followed by a Dirichlet/Hecke coefficient readout. Let $P_{D,N}$ be the projection onto $\Ran D_N$. The residual coefficient map is
\[
R_N=D_N^\dagger M_{R,N}.
\]
The coefficient visibility identity is
\[
R_N^*H_NR_N
=M_{R,N}^*P_{D,N}M_{R,N}
=G_{R,N}-E_{R,N}^*E_{R,N},
\]
where
\[
E_{R,N}=(I-P_{D,N})M_{R,N}.
\]
Thus the residual visibility constant is
\[
c_{R,N}=1-\|E_{R,N}G_{R,N}^{-1/2}\|^2.
\]
This is the residual-specific $\Xi$ adequacy gate: it measures what the residual sector sees that the coefficient dictionary cannot express.

\section{Source frame}
Let $\mathcal X_N$ be an upstream-declared character/source family. Define its coefficient-space Gram matrix by
\[
G_{\mathcal X,N}(m,n)
=\sum_{\omega\in\mathcal X_N}\lambda_{\omega,N}\omega(n)\overline{\omega(m)}.
\]
The induced residual source frame is
\[
F_{R,N}=R_N^*G_{\mathcal X,N}R_N.
\]

\begin{theorem}[Finite residual source absorption]
Assume
\[
G_{\mathcal X,N}\succeq \gamma_N H_N,
\qquad
R_N^*H_NR_N\succeq c_{R,N}G_{R,N}.
\]
Then
\[
F_{R,N}\succeq \gamma_Nc_{R,N}G_{R,N}.
\]
If additionally
\[
\Xi^{\BC}_N\preceq \sigma_NG_{R,N}+T_N^{\BC},
\]
then
\[
\Xi^{\BC}_N\preceq \frac{\sigma_N}{\gamma_Nc_{R,N}}F_{R,N}+T_N^{\BC}.
\]
In particular, the residual is source-absorbed when
\[
\gamma_Nc_{R,N}\to\infty,
\qquad
T_N^{\BC}\to0
\]
in the fixed/exhaustive ledger sense.
\end{theorem}

\begin{proof}
The first inequality follows by congruence:
\[
F_{R,N}=R_N^*G_{\mathcal X,N}R_N\succeq \gamma_NR_N^*H_NR_N\succeq \gamma_Nc_{R,N}G_{R,N}.
\]
Thus
\[
G_{R,N}\preceq (\gamma_Nc_{R,N})^{-1}F_{R,N},
\]
and substitution into the residual domination estimate gives the result.
\end{proof}

They cannot be selected because they make a final zero-confinement inequality pass. Zero simplicity is not an allowed assumption in this route.
