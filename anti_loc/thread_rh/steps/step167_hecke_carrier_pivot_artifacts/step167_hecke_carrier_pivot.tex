\section{Typed Hecke carrier declaration}

\[

be the unitary Hecke/Groessencharacter spectrum.  Finite-order narrow ray-class characters are allowed only as finite-window subfamilies of \(\X_K\).

\begin{definition}[Hecke \(L\)-function package]
For \(\chi\in\X_K\), let \(L_K(s,\chi)\) be the Hecke \(L\)-function.  The completed package is written
\[
  \Lambda_K(s,\chi)
  =
  A_K(\chi)^{s/2}L_{\infty,K}(s,\chi)L_K(s,\chi),
\]
with conductor/discriminant factor \(A_K(\chi)\), archimedean gamma factor \(L_{\infty,K}\), and the usual pole factors recorded separately for trivial characters.  The functional equation is
\[
  \Lambda_K(s,\chi)
  =
  \varepsilon(\chi)\Lambda_K(1-s,\overline{\chi}),
  \qquad |\varepsilon(\chi)|=1.
\]
For \(K=\Q\) and the trivial character \(1_\Q\), the scalar \(L\)-function is \(L_\Q(s,1_\Q)=\zeta(s)\).
\end{definition}

\begin{definition}[Hecke response carrier]
The Hecke carrier is the direct-integral anti-invariant response space
\[
  \Hh(K)=
  \int_{\X_K^-}^{\oplus}\mathcal H_{K,\chi}^{-}\,\dd\mu_K(\chi),
\]
where \(\mu_K\) is the Plancherel measure and the involution is induced by
\[
  (s,\chi)\mapsto (1-\overline{s},\overline{\chi}).
\]
The native probe family \(L_{\rm H}\) consists of upstream-declared Hecke explicit-formula features: prime-ideal/Euler probes, archimedean gamma probes, conductor and root-number probes, pole and trivial-character probes, tail/completion probes, and character readouts \(Q_\chi\).  The dissolving probe family \(D_{\rm H}\) consists of off-critical zero ledger readouts for \(\Lambda_K(s,\chi)\), plus the zeta-fiber descent readout when the \(K=\Q,\chi=1_\Q\) bridge is supplied.
\end{definition}

\section{The Hecke residual}

\begin{definition}[Hecke adequacy residual]
Let \(C_{\rm H}\) be the Hecke audit form on the legal quotient of \(\Hh(K)\).  Define
\[
  K^{\rm H}_{DD}=D_{\rm H}C_{\rm H}^{\dagger}D_{\rm H}^*,\quad
  K^{\rm H}_{DL}=D_{\rm H}C_{\rm H}^{\dagger}L_{\rm H}^*,\quad
  K^{\rm H}_{LL}=L_{\rm H}C_{\rm H}^{\dagger}L_{\rm H}^*.
\]
The Hecke-side residual is
\[
  \boxed{
  \XiH
  =
  K^{\rm H}_{DD}
  -
  K^{\rm H}_{DL}(K^{\rm H}_{LL})^\dagger K^{\rm H}_{LD}.
  }
\]
\end{definition}

\begin{proposition}[Fiberwise form, conditional]
If \(C_{\rm H}\), \(L_{\rm H}\), and \(D_{\rm H}\) are decomposable over \(\chi\), and if domains and pseudoinverses are compatible with the direct integral, then
\[
  \XiH
  =
  \int_{\X_K^-}^{\oplus}
  \Xi_{K,\chi}\,\dd\mu_K(\chi),
\]
where
\[
  \Xi_{K,\chi}
  =
  K_{DD}^{K,\chi}
  -
  K_{DL}^{K,\chi}(K_{LL}^{K,\chi})^\dagger K_{LD}^{K,\chi}.
\]
For finite character windows this becomes a finite direct sum only on the finite carrier and has status \texttt{moving\_window\_support\_only} unless a fixed completed ledger and tail record are supplied.
\end{proposition}

\section{Bridge to the Burnol/Sonine residual}

The scalar identity \(L_\Q(s,1_\Q)=\zeta(s)\) is accepted at the \(L\)-function level.  It does not identify the Schur carriers.  Equality or transfer between \(\XiBC\) and the \(K=\Q,\chi=1_\Q\) Hecke fiber requires maps
\[
  B_{\rm HB}:\mathcal H_{\rm BC}\to \mathcal H_{\Q,1_\Q},
  \qquad
  R_{\rm BH}:\mathcal H_{\Q,1_\Q}\to \mathcal H_{\rm BC},
\]
with an energy comparison, native-probe compatibility, dissolving-probe compatibility, and a bridge defect ledger.  The target shape is
\[
  K_{\rm BC}\preceq R_{\rm BH}K_{\rm H,Q,1}R_{\rm BH}^*+E_{\rm br},
\]
or an exact equality as a special case.  Inherited records do not supply these maps.  Therefore the bridge status is \texttt{open\_external\_bridge}.

\section{Initial residual branches}

The natural Hecke cascade is not a copy of the Burnol/Sonine cascade.  It has analogues of Calkin bridging and finite-carrier diagnostics, but its native first axes are character fibers, Plancherel tails, and auxiliary \(L\)-function explicit-formula records.

\begin{enumerate}[label=\textbf{H\arabic*.}]
\item \textbf{Direct-integral Schur branch.}  Audit decomposability and pseudoinverse compatibility for the formula \(\XiH=\int^\oplus\Xi_{K,\chi}\dd\mu_K\).
\item \textbf{Source lower-frame / Plancherel-tail branch.}  Audit \(F_n=U^*M_{\lambda_n}U\) and whether \(\lambda_n(\chi)\ge \Lambda_n\) holds on a fixed or exhaustive character ledger.
\item \textbf{Hecke Calkin bridge branch.}  Build the Hecke completed response/Calkin algebra and reject public-shadow substitutions unless a faithful bridge is supplied.
\item \textbf{Per-character finite-carrier branch.}  Run finite \(K,\chi,\rho\) packets only as finite-carrier diagnostics.
\item \textbf{Auxiliary explicit-formula branch.}  Supply conductor, gamma, root-number, pole, primitive/imprimitive, and tail records for \(L(s,\chi)\).
\item \textbf{Zeta-fiber descent branch.}  Audit the \(K=\Q,\chi=1_\Q\) bridge to the Burnol/Sonine pulled-evaluator residual.
\end{enumerate}
