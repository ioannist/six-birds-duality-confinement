\section{Purpose}
Steps 130--136 showed that the unrestricted BPRZ shifted-main-term kernel cannot be imported as a full source lower frame on the whole coefficient space: squarefree Boolean near-null modes form a GCD-log blind sector. Steps 132--135 gave a zeta/Muentz incidence-convolution repair, but Step 136 corrected the metric: in the source-compatible central-line Dirichlet metric the raw incidence operator is not the unweighted Boolean zeta matrix. This step therefore audits the sharper object
\[
  \bigl\|\Pi_{\B} Z_y (I-P_{\Omega\le K}) B_NG_{B,N}^{-1/2}\bigr\|,
\]
rather than the full incidence norm.

\section{Source-compatible Boolean model}
Let $P_y$ be a finite set of primes, $|P_y|=k$, and identify squarefree divisors with subsets $S\subseteq P_y$. In the central-line Dirichlet coefficient metric
\[
  \|a\|_H^2=\sum_n \frac{|a_n|^2}{n},
\]
the local normalized incidence block for a prime $p$ is
\[
  A_p=
  \begin{pmatrix}
  1&0\\ p^{-1/2}&1
  \end{pmatrix}.
\]
More generally, for a metric exponent $\alpha>0$, set $r_p=p^{-\alpha/2}$ and
\[
  A_{p,\alpha}=\begin{pmatrix}1&0\\ r_p&1\end{pmatrix}.
\]
The full incidence operator is
\[
  Z_{y,\alpha}=\bigotimes_{p\le y}A_{p,\alpha}.
\]
The full norm is
\[
  \|Z_{y,\alpha}\|=
  \prod_{p\le y}\sigma_+(p,\alpha),
  \qquad
  \sigma_+(p,\alpha)=\frac{\sqrt{4+p^{-\alpha}}+p^{-\alpha/2}}2.
\]
For $\alpha=1$ this is
\[
  \log\|Z_{y,1}\|=(1+o(1))\frac{\sqrt y}{\log y}.
\]

\section{Blind Walsh modes after the lawful metric}
Let
\[
  v_+=\frac1{\sqrt2}\binom{1}{1},
  \qquad
  v_- =\frac1{\sqrt2}\binom{1}{-1}.
\]
For a sign vector $\varepsilon\in\{\pm1\}^{P_y}$, write $A(\varepsilon)=\{p:\varepsilon_p=-1\}$. The Walsh blind projection is generated by tensor products
\[
  v_{\varepsilon}=\bigotimes_{p\le y}v_{\varepsilon_p}.
\]
The local output row after incidence is
\[
  \ell_{+,p}^{(\alpha)}=v_+^*A_{p,\alpha}
  =\frac1{\sqrt2}(1+r_p,\,1),
\]
\[
  \ell_{-,p}^{(\alpha)}=v_-^*A_{p,\alpha}
  =\frac1{\sqrt2}(1-r_p,\,-1).
\]
Thus
\[
  \langle v_\varepsilon,Z_{y,\alpha}b\rangle
  =\left(\bigotimes_{p\le y}\ell_{\varepsilon_p,p}^{(\alpha)}\right)b.
\]

\begin{warning}[metric-corrected leakage]
The exact upward-closure annihilation from the unweighted Boolean model is not literally true in the central-line metric. At $\alpha=0$, $r_p=1$ and $\ell_{-,p}^{(0)}=(0,-1)/\sqrt2$, so a negative coordinate can only read seed coefficients containing $p$. For $\alpha=1$, however,
\[
  \ell_{-,p}^{(1)}=\frac1{\sqrt2}(1-p^{-1/2},-1),
\]
so a negative coordinate also sees the absence of $p$, but with a leakage factor $1-p^{-1/2}$. Hence exact annihilation is replaced by weighted leakage control.
\end{warning}

\section{Blind-projected norms}
Let $\Pi_\B$ be the projection onto a declared blind Walsh family $\B\subseteq\{\pm1\}^{P_y}$. Let
\[
  P_{\Omega\le K}
  :=
  \text{orthogonal projection onto seed subsets }S\text{ with }|S|\le K.
\]
Define
\[
  L_{\B,K}(y,\alpha)
  :=\|\Pi_\B Z_{y,\alpha}P_{\Omega\le K}\|,
\]
\[
  T_{\B,K}(y,\alpha)
  :=\|\Pi_\B Z_{y,\alpha}(I-P_{\Omega\le K})\|.
\]
Then for any seed $b$,
\[
  \|\Pi_\B Z_{y,\alpha}b\|
  \le
  L_{\B,K}\|P_{\Omega\le K}b\|+T_{\B,K}\|(I-P_{\Omega\le K})b\|.
\]
The project-native residual is
\[
  \Xi_{{\rm GCD},N}
  =R_N^*\Pi_\B R_N,
  \qquad R_N=Z_{y,1}B_N.
\]
For the actual Burnol/Muentz seed window this gives
\[
  \Xi_{{\rm GCD},N}
  \preceq
  \delta_N^2G_{B,N},
\]
where a lawful upper bound is
\[
  \delta_N
  \le
  L_{\B,K}(y,1)
  +T_{\B,K}(y,1)
  \left\|(I-P_{\Omega\le K})B_NG_{B,N}^{-1/2}\right\|.
\]

\section{All-minus mode}
For the deepest all-minus Walsh mode $v_-^{\otimes k}$, the exact row norm is
\[
  \|v_-^{\otimes k}{}^*Z_{y,\alpha}\|^2
  =\prod_{p\le y}\left(1-p^{-\alpha/2}+\frac12p^{-\alpha}\right).
\]
For $\alpha=1$,
\[
  \log \|v_-^{\otimes k}{}^*Z_{y,1}\|
  =-(1+o(1))\frac{\sqrt y}{\log y}.
\]
Thus the deepest blind row is strongly suppressed in the source-compatible metric. This is a genuine improvement over the full norm, whose logarithm has the opposite sign at the same scale.

\section{Restricted BPRZ lower-frame salvage}
Suppose the GCD-log main kernel has lower frame
\[
  K_q\succeq \gamma_q I,
  \qquad \gamma_q\asymp\log q,
\]
on the nonblind sector. If
\[
  \|\Pi_\B R_NG_{B,N}^{-1/2}\|\le \delta_N<1,
\]
then
\[
  R_N^*K_qR_N
  \succeq
  \gamma_q(1-
\delta_N^2)G_{B,N}.
\]
Hence exact blind suppression is not required. A uniform floor $\delta_N<1$ is enough because $\gamma_q\to\infty$, provided finite-to-completed tail promotion is also earned.

\section{Status}
Step 137 does not prove the actual Burnol/Muentz residual seed has small blind projection. It proves the metric-correct algebraic audit:
\[
  \text{unweighted exact annihilation}
  \quad\leadsto\quad
  \text{source-compatible weighted leakage.}
\]
The next analytic record must estimate
\[
  L_{\B,K}(y,1)
  \quad\text{and}\quad
  T_{\B,K}(y,1)
  \left\|(I-P_{\Omega\le K})B_NG_{B,N}^{-1/2}\right\|
\]
for the actual residual windows. If this combined defect is $<1$ uniformly, the restricted BPRZ source route remains viable with a positive lower-frame floor. If it tends to zero, the GCD-log obstruction is removed on the actual residual class.
