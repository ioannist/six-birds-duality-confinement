\section{Purpose}
Steps 104--105 showed that raw shifted Sonin/prolate off-diagonal blocks
\[
  B_a=P_\infty\tau_a(I-P_\infty),\qquad a\ne0,
\]
carry a noncompact boundary-packet sector.  Steps 109--112 split the source route into
\[
  \Lambda_N=\gamma_Nc_N,
\]
where \(\gamma_N\) is character/source strength and \(c_N\) is coefficient visibility.  The key correction is that \(c_N\) is not a column-norm estimate for a fixed map.  It is a spanning residual for a constructible atom family.

This note attempts the Burnol density theorem needed for
\[
  \eps_N:=\|(I-P_{\A_N})M_NG_{\B,N}^{-1/2}\|\to0,
  \qquad
  c_N=1-\eps_N^2.
\]
The result is conditional but sharp: Burnol's co-Poisson machinery gives density inside the co-Poisson subspace, while the remaining load-bearing issue is whether the boundary-packet sector actually lies in that subspace.

\section{Burnol carrier records}
Let
\[
  K=L^2(0,\infty;dt)
\]
with cosine transform \(F_+\).  Burnol's extended Sonine space is
\[
  L_a=\{f\in K: f\text{ is constant on }(0,a),\;F_+f\text{ is constant on }(0,a)\},
\]
and the stricter Sonine space is
\[
  K_a=\{f\in K: f=0\text{ on }(0,a),\;F_+f=0\text{ on }(0,a)\}.
\]
For \(0<a<1\), set \(A=a^{-1}\).  The co-Poisson subspace \(P_a\subset L_a\) consists of square-integrable co-Poisson sums generated by functions \(g\in L^1(a,A;dt)\):
\[
  \C_a g(t)=\sum_{n\ge1}\frac{g(t/n)}{n}-\widehat g(1),
  \qquad
  \widehat g(s)=\int_0^\infty g(t)t^{-s}\,dt.
\]
Let \(Y_a\subset L_a\) be the closed span of the zeta-zero evaluator vectors \(Y^a_{\rho,k}\). Burnol's theorem gives, for \(a<1\),
\[
  \boxed{Y_a^\perp=P_a.}
\]
At \(a=1\), the zero-evaluator system is complete and minimal in \(L_1\). Thus Burnol's result is exactly a completed ledger/exhaustivity theorem for the zeta-zero readout.

\section{Boundary sector and the inclusion gate}
The boundary-packet sector from the semilocal prime-shift obstruction is
\[
  \B_S=
  \overline{\spanop}\{\Ran |P_\infty\tau_\ell(I-P_\infty)|:
  \ell\in \mathcal L_S\setminus\{0\}\},
\]
where
\[
  \mathcal L_S=\left\{\sum_{p\in S_f}k_p\log p:k_p\in\mathbb Z\right\}.
\]
To apply Burnol density directly, one must prove the inclusion gate
\[
  \boxed{\B_S\subseteq P_a.}
\]
Equivalently,
\[
  \boxed{\langle b,Y^a_{\rho,k}\rangle=0
  \quad\forall b\in\B_S,
  \forall\rho,
  0\le k<m_\rho.}
\]

\begin{definition}[Burnol quotient obstruction]
For a closed subspace \(\B\subset L_a\), define
\[
  \delta_{\B,a}:=\|(I-\Pi_{P_a})|_{\B}\|=\|\Pi_{Y_a}|_{\B}\|.
\]
Then \(\delta_{\B,a}=0\) if and only if \(\B\subseteq P_a\).
\end{definition}

\section{Declared atom family and density theorem}
Let \(\mathcal G_N\subset C_c^\infty([a,A])\) be a declared, non-smuggled generator family.  Define
\[
  \A_N=\spanop\{\C_ag:g\in\mathcal G_N\}\subset P_a,
\]
with pole/completion constants recorded if the Mellin vanishings \(\widehat g(0)=\widehat g(1)=0\) are not imposed.

\begin{theorem}[Burnol density transfer]
Assume:
\begin{enumerate}[label=(D\arabic*)]
\item \(0<a<1\) and Burnol's complement theorem \(P_a=Y_a^\perp\) holds.
\item The declared generator family is dense in the generator topology defining \(P_a\), and \(\C_a\) is continuous into \(L_a\).
\item The boundary sector satisfies \(\B\subseteq P_a\).
\end{enumerate}
Then
\[
  \overline{\bigcup_N\A_N}\supseteq \B.
\]
Consequently, for every finite boundary dictionary \(M_N:Y_{\B,N}\to L_a\) with \(\Ran M_N\subset\B\),
\[
  \eps_N=\|(I-P_{\A_N})M_NG_{\B,N}^{-1/2}\|\to0,
  \qquad
  c_N=1-\eps_N^2\to1.
\]
\end{theorem}

\begin{proof}
By Burnol's complement theorem, \(P_a\) is the closed co-Poisson complement of the zero-evaluator span.  By declared generator density, \(\overline{\bigcup_N\A_N}=P_a\).  If \(\B\subseteq P_a\), then every finite sub-dictionary of \(\B\) is approximated by strict atom extension.  The residual convergence follows from finite-dimensional approximation in an increasing dense family of subspaces.
\end{proof}

\begin{theorem}[Burnol quotient obstruction]
Let \(\A_N\subseteq P_a\) be any co-Poisson atom family. If \(\delta_{\B,a}>0\), then
\[
  \inf_N\|(I-P_{\A_N})|_{\B}\|\ge\delta_{\B,a}.
\]
In particular,
\[
  \liminf_N\eps_N\ge\delta_{\B,a}.
\]
\end{theorem}

\begin{proof}
For normalized \(b\in\B\), since \(\A_N\subseteq P_a\),
\[
  \|(I-P_{\A_N})b\|\ge \|(I-\Pi_{P_a})b\|=\|\Pi_{Y_a}b\|.
\]
Taking the supremum over \(b\in\B\) gives the result.
\end{proof}

\section{Consequence for the RH source route}
The active \(c_N\)-side question is now
\[
  \boxed{\delta_{\B,a}=0?}
\]
Equivalently, for each boundary packet \(b=P_\infty\tau_\ell(I-P_\infty)u\), its Burnol completed Mellin readouts must vanish at all nontrivial zeros:
\[
  M_b^{(k)}(\rho)=0,
  \qquad 0\le k<m_\rho.
\]

If this inclusion gate passes, Burnol density supplies \(c_N\to1\), and the source route reduces to the character/source strength \(\gamma_N\). If it fails, the nonzero \(Y_a\)-component is a coefficient-level \(\Xi\)-blind spot.

This step does not prove RH and does not prove \(\B\subseteq P_a\).  It proves the conditional density transfer and the exact quotient obstruction.  The next analytic test is
\[
  \Pi_{Y_a}P_\infty\tau_\ell(I-P_\infty)=0?
