\section{Boundary-symbol abstraction}
The hard-support model from Step 158 suggests that the noncompact part of $C_\ell$ is a boundary-leakage operator.  In the actual Burnol/Sonine carrier this must be declared as a typed bridge, not assumed.

\begin{definition}[Boundary-symbol factorization]
A boundary-symbol record for $C_\ell$ consists of a projection or positive contraction $W_\ell$ on the same Hilbert context, a bounded operator $U_\ell$, and a compact operator $K_\ell$ such that
\[
  C_\ell = U_\ell W_\ell + K_\ell.
\]
It is \emph{lower-faithful on $H_\eta$} if there exists $\alpha_\ell>0$ such that for every $u\in H_\eta$,
\[
  \|U_\ell W_\ell u\|
  \ge \alpha_\ell\|W_\ell u\|.
\]
The operator
\[
  \Xi^{\rm BW}_{\ell,a}:=P_\eta W_\ell^*W_\ell P_\eta
\]
is the boundary-Weyl sub-residual.
\end{definition}

\begin{theorem}[Boundary-Weyl avoidance dichotomy]
Assume a lower-faithful boundary-symbol factorization
\[
  C_\ell = U_\ell W_\ell+K_\ell,
  \qquad K_\ell\in\mathcal K(H).
\]
Then:
\begin{enumerate}[label=(\roman*)]
\item If $W_\ell P_\eta$ is compact, then $C_\ell P_\eta$ is compact.  Consequently $\Xi^{\rm BC}$ is norm-tail-payable under any finite-rank exhaustion of $H_\eta$.
\item If there exists a normalized weak-null sequence $u_n\in H_\eta$ such that
\[
  \liminf_{n\to\infty}\|W_\ell u_n\|\ge b>0,
\]
then
\[
  \|C_\ell P_\eta\|_{\rm e}\ge \alpha_\ell b>0.
\]
Consequently $\Xi^{\rm BC}$ is not compact/tail-payable by ordinary finite-window exhaustion.
\item Equivalently, compact/tail payment of $\Xi^{\rm BC}$ is reduced to the boundary avoidance condition
\[
  W_\ell P_\eta\in\mathcal K(H).
\]
\end{enumerate}
\end{theorem}

\begin{proof}
If $W_\ell P_\eta$ is compact, then $U_\ell W_\ell P_\eta$ is compact and so is $K_\ell P_\eta$. Hence $C_\ell P_\eta$ is compact.

Conversely, let $u_n\in H_\eta$ be normalized and weak-null with $\liminf\|W_\ell u_n\|\ge b$. Since $K_\ell$ is compact, $K_\ell u_n\to 0$. Thus
\[
  \liminf\|C_\ell u_n\|
  \ge
  \liminf\bigl(\|U_\ell W_\ell u_n\|-\|K_\ell u_n\|\bigr)
  \ge \alpha_\ell b.
\]
This is a Weyl-sequence certificate for positive essential norm.
\end{proof}

\section{Hard-support shadow}
In the hard-support shadow context,
\[
  P_+=\mathbf 1_{[0,\infty)},
  \qquad
  (T_{-\ell}f)(x)=f(x+\ell),
\]
and
\[
  C^{\rm supp}_\ell=(I-P_+)T_{-\ell}P_+.
\]
Then $C^{\rm supp}_\ell$ is a partial translation from $L^2([0,\ell])$ to $L^2([-\ell,0])$.  The boundary symbol is simply
\[
  W_\ell=\mathbf 1_{[0,\ell]}.
\]
Thus a sequence
\[
  u_n(x)=\ell^{-1/2}e^{2\pi i n x/\ell}\mathbf 1_{[0,\ell]}(x)
\]
is normalized, weak-null, and satisfies
\[
  \|C^{\rm supp}_\ell u_n\|=1.
\]
This proves noncompactness in the hard-support shadow.

\paragraph{Status.} This is \texttt{public\_shadow}, not a Burnol/Sonine theorem.  Promotion requires a Calkin-faithful bridge from $C_\ell$ to the boundary-symbol model in the pulled-evaluator context.
