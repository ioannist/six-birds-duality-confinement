\section{Purpose}
Steps 103--105 identified the noncompact semilocal obstruction as a boundary-packet sector generated by shifted Sonin/prolate off-diagonal blocks
\[
  B_\ell=P_\infty\tau_\ell(I-P_\infty),\qquad \ell\ne0.
\]
Steps 111--113 then reframed coefficient visibility as a Burnol/co-Poisson spanning problem rather than a fixed column-norm problem.  The present step tests the missing inclusion
\[
  \mathcal B\subseteq P_a,
\]
where $P_a$ is Burnol's co-Poisson complement in $L_a$.

\section{Burnol backbone}
Let $0<a<1$, $A=a^{-1}$, and let $L_a$ be the extended Sonine space consisting of functions constant on $(0,a)$ whose cosine transforms are also constant on $(0,a)$.  Let
\[
  Y_a=\overline{\operatorname{span}}
  \{Y^a_{\rho,k}: \rho\in Z(\zeta),\ 0\le k<m_\rho\}\subseteq L_a
\]
be the closed span of Burnol's zero-evaluator vectors.  Burnol's theorem states that for $a<1$ the perpendicular complement of $Y_a$ in $L_a$ is the co-Poisson subspace $P_a$.
Thus
\[
  P_a=Y_a^\perp.
\]
Equivalently, for $f\in L_a$,
\[
  f\in P_a
  \quad\Longleftrightarrow\quad
  M(f)^{(k)}(\rho)=0
  \quad\forall \rho,\\ 0\le k<m_\rho,
\]
where $M(f)$ is the completed or right-Mellin transform in Burnol's normalization.

\section{Boundary packet transport}
Let $J_a$ denote the declared transport from the archimedean/semilocal Sonin model into the Burnol $L_a$ carrier.  The transported boundary block is
\[
  \mathfrak B_\ell := J_aP_\infty\tau_\ell(I-P_\infty).
\]
The boundary-packet sector generated by finite-place log-shifts is
\[
  \mathcal B_a
  =\overline{\operatorname{span}}
  \{\operatorname{Ran}\mathfrak B_\ell:
    \ell\in\mathcal L_S\setminus\{0\}\}
  \subseteq L_a,
\]
provided the transport gate is lawful.

\begin{theorem}[Boundary-to-co-Poisson inclusion criterion]
Assume $\mathfrak B_\ell$ is a densely defined closed operator with range in $L_a$.  Then
\[
  \operatorname{Ran}\mathfrak B_\ell\subseteq P_a
\]
if and only if any, hence all, of the following equivalent conditions hold:
\begin{enumerate}[label=(\roman*)]
\item $\Pi_{Y_a}\mathfrak B_\ell=0$.
\item $\mathfrak B_\ell^*Y^a_{\rho,k}=0$ for every nontrivial zeta zero $\rho$ and $0\le k<m_\rho$.
\item $M(\mathfrak B_\ell u)^{(k)}(\rho)=0$ for every legal input $u$, every $\rho$, and every $0\le k<m_\rho$.
\end{enumerate}
Consequently,
\[
  \mathcal B_a\subseteq P_a
\]
if and only if these equivalent conditions hold for all finite-place shift modes $\ell\in\mathcal L_S\setminus\{0\}$.
\end{theorem}

\begin{proof}
Burnol's theorem gives $P_a=Y_a^\perp$.  Thus $\operatorname{Ran}\mathfrak B_\ell\subseteq P_a$ iff its projection to $Y_a$ vanishes, i.e. $\Pi_{Y_a}\mathfrak B_\ell=0$.  This is equivalent to the adjoint annihilation of the total zero-evaluator span.  Since the zero evaluators realize the functionals $f\mapsto M(f)^{(k)}(\rho)$, the final condition follows.
\end{proof}

\section{The zeta-factorization sufficient route}
Co-Poisson atoms earn orthogonality to the zeta-zero evaluators because their Mellin transforms contain a zeta factor.  Thus a strong sufficient condition for inclusion is the following.

\begin{proposition}[Zeta-factorization certificate]
Suppose that for every legal input $u$ there exists a function $\alpha_{\ell,u}$ in the appropriate Burnol generator class such that
\[
  M(\mathfrak B_\ell u)(s)=\zeta(s)\alpha_{\ell,u}(s),
\]
with the required trivial-zero, pole, support, and endpoint records.  Then
\[
  \operatorname{Ran}\mathfrak B_\ell\subseteq P_a.
\]
\end{proposition}

\begin{proof}
At every nontrivial zero $\rho$ of multiplicity $m_\rho$, the product $\zeta(s)\alpha_{\ell,u}(s)$ vanishes to order at least $m_\rho$, provided the product is legal in Burnol's function class.  Hence every zero-evaluator $Y^a_{\rho,k}$ annihilates $\mathfrak B_\ell u$.
\end{proof}

\begin{warning}[Raw shift non-certificate]
The raw shifted Sonin/prolate block $P_\infty\tau_\ell(I-P_\infty)$ does not by itself display a zeta factor.  It is a geometric/log-shift object.  Therefore it does not automatically land in $P_a$.  The inclusion must be earned either by zeta-factorization, by a direct zero-evaluator annihilation theorem, or by an explicit residual/source record.
\end{warning}

\section{Residual if inclusion fails}
Define the boundary-to-co-Poisson residual
\[
  \Xi^{\rm BC}_{\ell,a}
  :=\mathfrak B_\ell^*\Pi_{Y_a}\mathfrak B_\ell\succeq0.
\]
For a finite zero window $T$, let $Y_{a,T}$ be the span of $Y^a_{\rho,k}$ with $|\Im\rho|\le T$, and define
\[
  \Xi^{\rm BC}_{\ell,a,T}
  :=\mathfrak B_\ell^*\Pi_{Y_{a,T}}\mathfrak B_\ell.
\]
Then $\Xi^{\rm BC}_{\ell,a}=0$ is exactly the inclusion certificate.  If $\Xi^{\rm BC}_{\ell,a}\ne0$, this is a genuine $\Xi$-style adequacy residual: it is the part of the boundary-packet sector visible to zero evaluators but invisible to the Burnol/co-Poisson atom family.

\begin{theorem}[Finite-window promotion gate]
Let $G_{\mathcal B,N}$ be a finite boundary Gram form and $M_N$ a finite boundary dictionary.  If
\[
  \|\Pi_{Y_{a,T}}M_NG_{\mathcal B,N}^{-1/2}\|\to0
\]
for an exhausting zero window $T=T_N$, and if the zero-ledger tail satisfies
\[
  \|(\Pi_{Y_a}-\Pi_{Y_{a,T_N}})M_NG_{\mathcal B,N}^{-1/2}\|\to0,
\]
then the finite Burnol/co-Poisson visibility constants satisfy $c_N\to1$.
Without the second tail condition, finite-window annihilation is only support evidence.
\end{theorem}

\section{Decision}
Step 114 does not prove $\mathcal B_a\subseteq P_a$.  It proves that this inclusion is equivalent to zero-evaluator annihilation of the shifted boundary packets, and it identifies the exact residual if annihilation fails:
\[
  \Xi^{\rm BC}_{a}=\sum_{\ell}\mathfrak B_\ell^*\Pi_{Y_a}\mathfrak B_\ell.
\]
Thus the next load-bearing theorem is either:
\[
  M(J_aP_\infty\tau_\ell(I-P_\infty)u)(s)=\zeta(s)\alpha_{\ell,u}(s),
\]
or source absorption of the residual:
\[
  \Xi^{\rm BC}_{a}\preceq F_n+E_n,
  \qquad F_n\succeq \Lambda_n(\Theta_0^-)^{-1},\quad \Lambda_n\to\infty.
\]
