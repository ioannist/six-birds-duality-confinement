\section{Purpose}
Step 99 stated the abstract promotion gate from finite character windows to a completed lower frame.  Step 100 instantiates that gate on one concrete carrier: the semilocal Connes--Consani--Moscovici response space
\[
  \Y_S=L^2(\R,dm_S),\qquad
  dm_S(\xi)=\left|\prod_{v\in S}L_v\left(\frac12-i\xi\right)\right|^2d\xi,
\]
for a finite set of places $S$ containing $\infty$.  This is the Tier-1 hybrid route: keep the paired Weil/membrane gates, import archimedean Sonin/prolate positivity, and formulate the finite-prime/character-source absorption problem inside the semilocal response space.

This note does \emph{not} prove RH.  It fixes the exact objects that a semilocal Plancherel/tail theorem must control.

\section{Semilocal response carrier}
Let $S$ be a finite set of places with $\infty\in S$ and finite part $S_f$.  Write
\[
  E_S(\xi):=\prod_{v\in S}L_v\left(\frac12-i\xi\right),
  \qquad dm_S(\xi):=|E_S(\xi)|^2d\xi.
\]
The semilocal Hardy--Titchmarsh transform gives a unitary model
\[
  V_S=M_SU_S:L^2(X_S)^{K_S}\longrightarrow L^2(\R,dm_S),
\]
where finite local factors enter through
\[
  F_\mu w_S(\eta_S f)(\xi)=\prod_{p\in S_f}L_p\left(\frac12-i\xi\right)F_\mu w_\infty f(\xi).
\]
The functional-equation grading becomes the symmetry $\xi\mapsto-\xi$ under this transform.  Hence we define the anti-invariant response space
\[
  \Y_S^-:=\{y\in L^2(\R,dm_S):J_Sy=-y\},
  \qquad (J_Sy)(\xi)=\overline{y(-\xi)}.
\]

\begin{definition}[Semilocal window maps]
A semilocal finite window is a positive contraction
\[
  P_{S,T,N}^-:\Y_S^-\to\Y_S^-
\]
constructed from two records:
\begin{enumerate}[label=(\roman*)]
\item a spectral cutoff $P_Ty=1_{[-T,T]}y$ in $L^2(\R,dm_S)$;
\item a finite-rank cyclic-pair/prolate/orthogonal-polynomial projection $P_N^S$ associated to the cyclic pair $(S,\xi_S)$ in the semilocal Hardy--Titchmarsh model.
\end{enumerate}
When $P_T$ and $P_N^S$ fail to commute, $P_{S,T,N}^-$ is defined as the orthogonal projection onto the anti-invariant finite-dimensional span generated by $P_TP_N^S\Y_S^-$, not as an illicit commuting product.
The analysis map is
\[
  \Pi_{S,T,N}:\Y_S^-\to \Y_{S,T,N}^-:=\Ran P_{S,T,N}^-.
\]
\end{definition}

\begin{warning}[No commuting-window overread]
The notation $P_TP_N^S$ is only a generator for the finite span.  Unless a commutator defect record is supplied, one may not treat spectral cutoff, prolate/cyclic-pair cutoff, and conductor-place cutoff as mutually commuting projections.
\end{warning}

\section{Tail records}
Let $(\Theta_{0,S}^-)^{-1}$ be the completed inverse budget form on $\Y_S^-$.  A finite window is useful only with a tail form $\Ttail_{S,T,N}\succeq0$ satisfying
\begin{equation}\label{eq:tail-form}
  (\Theta_{0,S}^-)^{-1}
  \preceq
  \Pi_{S,T,N}^{*}B_{S,T,N}\Pi_{S,T,N}
  +\Ttail_{S,T,N},
\end{equation}
where $B_{S,T,N}$ is the inverse-budget form actually used on the finite window.

If $\mathsf A_Z$ is the completed zero-side anti-invariant ledger, the ledger-tail observable is
\[
  \tau_Z(S,T,N):=\\Tr\bigl((I-P_{S,T,N}^-)\mathsf A_Z(I-P_{S,T,N}^-)\bigr).
\]
The required promotion condition is not $\|\Ttail_{S,T,N}\|\to0$ in operator norm.  It is a fixed/exhaustive ledger condition such as
\[
  \tau_Z(S,T,N)\to0,
\]
plus the corresponding source and bridge defects vanishing.

\begin{lemma}[Trace-class ledger exhaustivity]
Let $\mathsf A_Z\succeq0$ be trace class on $\Y_S^-$ and let $P_n\to I$ strongly be an increasing sequence of finite-rank projections. Then
\[
  \Tr\bigl((I-P_n)\mathsf A_Z(I-P_n)\bigr)\to0.
\]
Thus a finite-window ladder can promote to the completed zero ledger only through a trace-class or otherwise audited tail record.
\end{lemma}

\begin{proof}
For finite-rank positive $A_m\uparrow \mathsf A_Z$ in trace norm, choose $m$ so that $\|\mathsf A_Z-A_m\|_1<\epsilon$.  Since $P_n\to I$ strongly and $A_m$ has finite rank, $(I-P_n)A_m(I-P_n)\to0$ in trace norm.  The remainder is bounded by $\epsilon$ in trace norm.  Letting $n\to\infty$ and then $\epsilon\to0$ gives the claim.
\end{proof}

\section{Finite-window lower frame and completed promotion}
Let the finite-window character/source frame be
\[
  \F^{\rm win}_{S,T,N}
  =\sum_{\omega\in\mathcal X_{S,T,N}}
  \lambda_{\omega,S,T,N}Q_{\omega}^{*}\Theta_{\omega}^{-1}Q_{\omega}
  \quad\text{on }\Y_{S,T,N}^-.
\]
Assume the finite-window lower frame
\begin{equation}\label{eq:finite-lower-frame}
  \F^{\rm win}_{S,T,N}\succeq \Lambda_{S,T,N}B_{S,T,N}.
\end{equation}
Lift it to the completed space by
\[
  \F_{S,T,N}:=\Pi_{S,T,N}^{*}\F^{\rm win}_{S,T,N}\Pi_{S,T,N}.
\]
Combining \eqref{eq:tail-form} and \eqref{eq:finite-lower-frame} gives
\begin{equation}\label{eq:completed-lower-frame-up-to-tail}
  \F_{S,T,N}+\Lambda_{S,T,N}\Ttail_{S,T,N}
  \succeq
  \Lambda_{S,T,N}(\Theta_{0,S}^-)^{-1}.
\end{equation}
Equivalently,
\[
  \F_{S,T,N}
  \succeq
  \Lambda_{S,T,N}(\Theta_{0,S}^-)^{-1}
  -\Lambda_{S,T,N}\Ttail_{S,T,N}.
\]

\begin{theorem}[Semilocal source-frame promotion]
Suppose that along a directed ladder $(S,T,N)$ we have:
\begin{enumerate}[label=(\alph*)]
\item finite-window lower frame \eqref{eq:finite-lower-frame};
\item inverse-budget tail record \eqref{eq:tail-form};
\item $\Lambda_{S,T,N}\to\infty$;
\item the tail contribution $\Lambda_{S,T,N}\Ttail_{S,T,N}$ is harmless on the completed zero ledger, e.g. it has vanishing ledger trace or is absorbed by a fixed/exhaustive tail bridge;
\item source and bridge defects vanish.
\end{enumerate}
Then the completed anti-invariant source currency collapses in the sense required by the fixed/exhaustive ledger theorem.
\end{theorem}

The hybrid route's active cross-term is

  \Delta_S^+\preceq \F_{S,T,N}+E_{\rm abs,S,T,N}.

\begin{theorem}[Semilocal carrier endpoint]
If \eqref{eq:delta-absorb} holds, the completed lower-frame promotion holds, $\Xi_C(D_{\rm off}\mid L_S^-)$ is controlled, and the zero ledger is fixed or exhaustive, then the semilocal hybrid route yields the framework-level chain
\[
  \mathsf A_Z\preceq \mathsf K_{\rm cmp,S}\preceq \Theta_S^-+E_{S,T,N},
  \qquad \Tr E_{S,T,N}\to0,

\end{warning}

\section{Compact $\Delta_S$ shortcut}
The user's strategic note correctly identifies compactness of $\Delta_S$ as the highest-leverage test.

\begin{proposition}[Compact cross-term shortcut]
Assume $\Delta_S^+$ is compact on $\Y_S^-$ and $P_n\to I$ strongly is an exhaustive window ladder.  Then for every $\epsilon>0$ there exists $n_0$ and a finite-rank positive operator $D_{n_0}$ supported on $P_{n_0}\Y_S^-$ such that
\[
  \Delta_S^+\preceq D_{n_0}+\epsilon I
\]
as a form on the unit ball of the completed response norm.  If the finite-window source frame dominates $D_{n_0}$ and the residual $\epsilon I$ is absorbed by the lower-frame/tail budget, then \eqref{eq:delta-absorb} follows.
\end{proposition}

\begin{proof}
Compact positive operators are norm limits of finite-rank positive compressions.  Choose $n_0$ so that $\|(I-P_{n_0})\Delta_S^+(I-P_{n_0})\|<\epsilon$ and take $D_{n_0}=P_{n_0}\Delta_S^+P_{n_0}$, with cross-terms absorbed by the usual $2ab\le t a^2+t^{-1}b^2$ form estimate if $P_{n_0}$ does not reduce $\Delta_S^+$.
\end{proof}

If this compactness theorem fails for the actual semilocal $\Delta_S$, the failure mode identifies the non-compact anti-invariant sector that must be covered by the Hecke/Dirichlet source frame.
