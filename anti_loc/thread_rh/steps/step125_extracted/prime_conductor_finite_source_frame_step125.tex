\section{Finite prime-conductor model}
Let $q$ be prime and let
\[
\mathcal N\subset\{1,\dots,L\},\qquad L<q,
\]
be the finite coefficient support produced by the regularized Burnol-to-Dirichlet shadow.  For a coefficient vector $a=(a_n)_{n\in\mathcal N}$ define
\[
Q_\chi a=\sum_{n\in\mathcal N}a_n\chi(n).
\]
The full character Gram is
\[
G_q(m,n)=\sum_{\chi\bmod q}\chi(n)\overline{\chi(m)}.
\]
Since $m,n<q$, the congruence $m\equiv n\pmod q$ is equivalent to equality.  Hence
\[
G_q=(q-1)I_{\mathcal N}.
\]
This is the exact finite lower frame.

\begin{theorem}[Complete prime-modulus finite lower frame]
If $q$ is prime and $\mathcal N\subset\{1,\dots,L\}$ with $L<q$, then
\[
\sum_{\chi\bmod q}\left|\sum_{n\in\mathcal N}a_n\chi(n)\right|^2
=(q-1)\sum_{n\in\mathcal N}|a_n|^2.
\]
Thus $G_q\succeq(q-1)I_{\mathcal N}$ exactly.
\end{theorem}

\section{Primitive/nonprincipal correction}
For $q$ prime, the primitive characters modulo $q$ are precisely the nonprincipal characters.  Removing the principal character gives
\[
G_q^{\rm prim}=(q-1)I_{\mathcal N}-\mathbf 1\mathbf 1^*.
\]
If $M=|\mathcal N|$, then the eigenvalues are
\[
q-1\quad(M-1\text{ times}),
\qquad q-1-M\quad(1\text{ time}).
\]
Thus the primitive/nonprincipal family has a finite lower frame iff $M<q-1$.

\begin{theorem}[Prime primitive correction]
For $q$ prime and $\mathcal N\subset\{1,
\dots,L\}$ with $M=|\mathcal N|<q-1$,
\[
\sum_{\substack{\chi\bmod q\\ \chi\ne\chi_0}}
\left|\sum_{n\in\mathcal N}a_n\chi(n)\right|^2
\ge (q-1-M)\sum_{n\in\mathcal N}|a_n|^2.
\]
The removed principal character is therefore a rank-one defect.
\end{theorem}
