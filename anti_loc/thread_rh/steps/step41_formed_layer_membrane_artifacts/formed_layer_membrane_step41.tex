\begin{abstract}
This note assembles the previous anti-localization results into one formed-layer membrane theorem.  A formed closure/layer is not claimed to be needle-free in an absolute substrate sense.  It is claimed to have a membrane exactly when its declared native predictive probe family has a common currency budget and its native critical-pair witness ledger is complete.  Equivalently, every native recombination probe has finite audited shadow price under the exact package, at every accepted predictive stage, and every failure witness has been priced, repaired, or lawfully excluded.  The theorem is closure-scoped: non-closed artifacts are outside the claim.
\end{abstract}

\section{Purpose and Scope}
The previous notes established the scalar and matrix capacity identities, predictive family capacity, admissible test-family gates, critical-pair witnesses, strict-extension repair, and root-composite obstruction ledgers.  This step extracts the common theorem:
\[
\boxed{\text{formed closure}+\text{accepted predictive native test family}+\text{completed witness ledger}}
\]
\[
\boxed{\Longleftrightarrow\quad \text{layer membrane/no unresolved native needles}.}
\]
The theorem is not a universal statement that raw substrates have no needles.  A raw or unclosed artifact may have needles.  The claim is only that a formed closure/layer with an accepted native anti-localization record has no \emph{native unresolved predictive recombination needle}.

\section{Predictive Closure Records}

\begin{definition}[Exact packaged stage]
At predictive stage $j$, an exact packaged carrier consists of a finite-dimensional Hilbert coordinate space $E_j$, a Hilbert carrier $H_j$, a package map
\[
  \Gamma_j:E_j\to H_j,
\]
a positive semidefinite audit form $C_j\succeq0$ on $H_j$, and a declared native probe family
\[
  L_j:H_j\to Y_j.
\]
The packaged audit and packaged probe are
\[
  C_{\Gamma,j}:=\Gamma_j^*C_j\Gamma_j,
  \qquad
  L_{\Gamma,j}:=L_j\Gamma_j.
\]
A response transport
\[
  R_j:Y_j\to Y
\]
places all responses into a common predictive response space $Y$.
\end{definition}

\begin{definition}[Null-mode legality]
Stage $j$ is null-legal for $L_j$ when
\[
  L_{\Gamma,j}a=0\qquad (a\in\Null C_{\Gamma,j}).
\]
Equivalently, no declared native probe sees a zero-audit direction.  If null-mode legality fails, the variational capacity of some native recombination is infinite, even if a pseudoinverse expression hides the defect.
\end{definition}

\begin{definition}[Transported probe-family currency]
At a null-legal stage $j$, the transported family currency matrix is
\[
  \widehat K_j
  :=R_jL_{\Gamma,j}C_{\Gamma,j}^{\dagger}L_{\Gamma,j}^*R_j^*\succeq0.
\]
For $y\in Y$, the transported recombination probe is
\[
  \ell_{j,y}(u)=\langle y,R_jL_j u\rangle_Y.
\]
\end{definition}

\begin{lemma}[Currency represents all native recombinations]
Assume stage $j$ is null-legal.  Then for every $y\in Y$,
\[
  \Capacity_{C_j}(\ell_{j,y};\Range\Gamma_j)=y^*\widehat K_j y.
\]
\end{lemma}

\begin{proof}
The recombination probe has packaged representative
\[
  g_{j,y}=L_{\Gamma,j}^*R_j^*y\in E_j.
\]
By the scalar capacity identity on the exact package,
\[
\Capacity_{C_j}(\ell_{j,y};\Range\Gamma_j)
  =g_{j,y}^*C_{\Gamma,j}^{\dagger}g_{j,y}
  =y^*R_jL_{\Gamma,j}C_{\Gamma,j}^{\dagger}L_{\Gamma,j}^*R_j^*y.
\]
\end{proof}

\begin{definition}[Predictive budget]
A predictive family budget is a positive semidefinite form
\[
  \Theta\succeq0
\]
on $Y$.  Stage $j$ passes the budget when
\[
  \widehat K_j\preceq\Theta.
\]
The predictive currency budget passes when
\[
  \widehat K_j\preceq\Theta
  \qquad\text{for every accepted predictive stage }j.
\]
\end{definition}

\section{Native Needles and Critical-Pair Ledgers}

\begin{definition}[Native predictive recombination needle]
A native predictive recombination needle is a pair $(j,y)$ such that stage $j$ is accepted in the predictive closure record and
\[
  y^*(\widehat K_j-\Theta)y>0.
\]
It is a \emph{critical-pair witness} when its support is minimal relative to a declared finite probe coordinate/frame.
\end{definition}

\begin{definition}[Witness ledger]
A witness ledger $\mathcal W$ records every native witness as one of the following statuses:
\[
\texttt{priced},\quad
\texttt{carrier\_repaired},\quad
\texttt{family\_completed},\quad
\texttt{excluded\_with\_nonclaim},\quad
\texttt{downgraded},\quad
\texttt{unresolved}.
\]
The ledger is complete for the budget $\Theta$ when no accepted predictive stage contains an unresolved witness.
\end{definition}

\begin{definition}[Accepted predictive native test family]
A predictive family record
\[
  \mathfrak T=(\{E_j,H_j,\Gamma_j,C_j,L_j,R_j\}_j,Y,\Theta,\mathcal W,\mathsf A)
\]
is accepted when the following gates pass:
\begin{enumerate}[label=(G\arabic*)]
\item formed closure/layer status is active;
\item package exactness and null-mode legality hold at all accepted stages;
\item $L_j$ is a declared native probe family and is recombination-closed after response transport;
\item protocol/route union and readout-agreement defects are inside the declared budget;
\item predictive transport is lawful and the common budget $\Theta$ is visible;
\item all six role channels have declared statuses;
\item no post-hoc selection, hidden probe, public-shadow overread, or finite-to-exact smuggling is used;
\item the witness ledger is complete or the claim is explicitly downgraded.
\end{enumerate}
\end{definition}

\section{The Formed-Layer Membrane Theorem}

\begin{definition}[Layer membrane]
A formed closure/layer has an accepted anti-localization membrane relative to $(\mathfrak T,\Theta)$ when all native predictive recombination attempts are either priced by $\Theta$ or lawfully excluded from the claim, and no unresolved native witness remains.
\end{definition}

\begin{theorem}[Formed-layer membrane theorem]
Let $\mathfrak T$ be a formed predictive closure record with exact packaged stages, null-mode legality, declared native probe families, lawful response transports, and a complete no-smuggling audit record.  Then the following are equivalent.
\begin{enumerate}[label=(\roman*)]
\item $\mathfrak T$ has an accepted anti-localization membrane relative to budget $\Theta$.
\item The predictive family currency budget passes:
\[
  \widehat K_j\preceq\Theta
  \qquad\text{for every accepted predictive stage }j,
\]
and every non-budgeted recombination has an explicit nonclaim/downgrade record.
\item There is no unresolved native predictive critical-pair witness:
\[
  y^*(\widehat K_j-\Theta)y\le0
  \qquad\text{for every accepted }j\text{ and every native recombination }y.
\]
\end{enumerate}
Consequently, for every native recombination $y$,
\[
  \sup_j \Capacity_{C_j}(\ell_{j,y};\Range\Gamma_j)
  \le y^*\Theta y.
\]
\end{theorem}

\begin{proof}
By the currency representation lemma,
\[
  \Capacity_{C_j}(\ell_{j,y};\Range\Gamma_j)=y^*\widehat K_jy.
\]
For a Hermitian matrix $D_j=\widehat K_j-\Theta$, Loewner negativity
\[
  D_j\preceq0
\]
is equivalent to
\[
  y^*D_jy\le0\qquad\forall y.
\]
Thus the budget inequality $\widehat K_j\preceq\Theta$ for all accepted $j$ is exactly the assertion that no accepted stage contains a positive native recombination witness.  If a recombination is excluded by a declared nonclaim, it is outside the membrane claim; otherwise a positive witness is unresolved and the membrane status cannot be accepted.  This proves equivalence of (ii) and (iii), and (i) is precisely the status interpretation of those algebraic conditions under the accepted gate list.  The capacity bound follows by taking the supremum over $j$.
\end{proof}

\begin{corollary}[Finite eigenvalue certificate]
Assume $Y$ is finite-dimensional and $\Theta\succ0$.  The membrane budget is equivalent to
\[
  \lambda_{\max}\bigl(\Theta^{-1/2}\widehat K_j\Theta^{-1/2}\bigr)\le1
\]
for every accepted predictive stage $j$.  A failing eigenvector gives a canonical native critical-pair witness.
\end{corollary}

\begin{proof}
This is the standard spectral characterization of Loewner order after congruence by $\Theta^{-1/2}$.  If the largest eigenvalue exceeds $1$, the corresponding eigenvector $y$ satisfies $y^*(\widehat K_j-\Theta)y>0$.
\end{proof}

\begin{corollary}[Summable-defect membrane propagation]
Suppose transported currencies satisfy
\[
  \widehat K_{j+1}\preceq(1+\varepsilon_j)\widehat K_j+D_j,
\]
with
\[
  \prod_j(1+\varepsilon_j)\le P<\infty,
  \qquad
  \sum_jD_j\preceq D_\infty.
\]
If
\[
  P(\widehat K_0+D_\infty)\preceq\Theta,
\]
then the predictive membrane budget passes at every accepted stage.
\end{corollary}

\begin{proof}
Iterating the recurrence gives the standard discrete Gronwall bound
\[
  \widehat K_n\preceq P(\widehat K_0+D_\infty).
\]
The assumed Loewner budget then implies $\widehat K_n\preceq\Theta$ for all $n$.
\end{proof}

\begin{corollary}[Closure-only scope]
If the formed-closure gate fails, the membrane theorem does not apply.  A raw substrate, provisional package, unaccepted protocol stack, or undeclared probe family may contain needles without contradicting the theorem.
\end{corollary}

\section{Interpretation}
In currency language, $\widehat K_j$ is the table of shadow prices for all native recombination probes at predictive stage $j$.  In membrane language, $\widehat K_j\preceq\Theta$ says that every native way of addressing the formed layer has a declared finite price.  In completion language, a positive eigenvector of $\widehat K_j-\Theta$ is an unresolved critical pair.  Therefore a layer membrane is not a scalar estimate: it is a completed predictive family ledger.

\section{Nonclaims}
The theorem does not assert any of the following:
\begin{enumerate}[label=(N\arabic*)]
\item all raw substrates have no needles;
\item passivity alone gives anti-localization;
\item scalar/diagonal probe checks imply family anti-localization;
\item finite-stage anti-localization implies predictive anti-localization;
\item public shadows or ambient diagonal integrals are lawful witnesses without a bridge;
\item RH or Navier-Stokes are solved by the framework theorem.
\end{enumerate}
