\section{Burnol carrier and co-Poisson synthesis}

Let
\[
K=L^2(0,\infty;dt)
\]
with cosine transform \(F_+\).  Burnol's extended Sonine space \(L_a\) consists of functions
which are constant on \((0,a)\) and whose cosine transform is also constant on \((0,a)\).  The
strict Sonine space \(K_a\) consists of functions which vanish on \((0,a)\) and whose cosine
transform vanishes on \((0,a)\).

For \(0<a<1\), set \(A=a^{-1}\).  For a generator \(g\) supported in \([a,A]\), define its right
Mellin transform by
\[
\widehat g(s)=\int_0^\infty g(t)t^{-s}\,dt.
\]
The co-Poisson synthesis is
\[
\mathcal C_a g(x)
=
\sum_{n\ge 1}\frac{g(x/n)}{n}
-
\widehat g(1).
\]
The dual side is
\[
\mathcal C_a^\vee g(t)
=
\sum_{m\ge 1}\frac{g(m/t)}{t}
-
\widehat g(0),
\]
and Burnol's co-Poisson identity gives
\[
F_+(\mathcal C_a^\vee g)=\mathcal C_a g.
\]
Thus if
\[
\widehat g(0)=0,\qquad \widehat g(1)=0,
\]
then \(\mathcal C_a g\in K_a\).

\section{Declared atom family}

\begin{definition}[Non-smuggled Burnol/co-Poisson atoms]
Fix \(0<a<1\), \(A=a^{-1}\), a deterministic nested partition
\[
\mathcal P_N=\{I_{\nu,N}\}_\nu
\]
of \([a,A]\), and two declared correction atoms \(\kappa_0,\kappa_1\in C_c^\infty(a,A)\)
whose moment matrix
\[
M_\kappa=
\begin{pmatrix}
\int \kappa_0(t)\,dt & \int \kappa_1(t)\,dt\\
\int \kappa_0(t)t^{-1}\,dt & \int \kappa_1(t)t^{-1}\,dt
\end{pmatrix}
\]
is invertible.
For each raw bump \(\beta_{\nu,N}\in C_c^\infty(I_{\nu,N})\), define
\[
g_{\nu,N}
=
\beta_{\nu,N}
-
(\kappa_0,\kappa_1)M_\kappa^{-1}
\begin{pmatrix}
\int \beta_{\nu,N}(t)\,dt\\
\int \beta_{\nu,N}(t)t^{-1}\,dt
\end{pmatrix}.
\]
Then
\[
\widehat g_{\nu,N}(0)=\widehat g_{\nu,N}(1)=0.
\]
The Burnol/co-Poisson atom is
\[
\phi_{\nu,N}=\mathcal C_a g_{\nu,N}.
\]
\end{definition}

The family is non-smuggled because the support scale, partition, correction atoms, and moment
conditions are declared before seeing the target residual or zero locations.

\section{Visibility is a residual}

Let \(M_N:Y_{\mathcal B,N}\to H_N\) synthesize a finite boundary-packet dictionary and let
\[
G_{\mathcal B,N}=M_N^*M_N.
\]
Let \(D_N\) synthesize the declared atom dictionary
\[
D_N:\mathbb C^{I_N}\to H_N,\qquad e_\nu\mapsto \phi_{\nu,N}.
\]
Let \(P_N\) be the orthogonal projection onto \(\operatorname{Ran}D_N\).  Define
\[
E_{\mathrm{coef},N}=(I-P_N)M_N.
\]

\begin{theorem}[Finite visibility identity]
With
\[
R_N=D_N^\dagger M_N,\qquad H_N=D_N^*D_N,
\]
one has
\[
R_N^*H_NR_N
=
M_N^*P_NM_N
=
G_{\mathcal B,N}-E_{\mathrm{coef},N}^*E_{\mathrm{coef},N}.
\]
Consequently,
\[
c_N
=
\lambda_{\min}\!\left(
G_{\mathcal B,N}^{-1/2}R_N^*H_NR_NG_{\mathcal B,N}^{-1/2}
\right)
=
1-\left\|E_{\mathrm{coef},N}G_{\mathcal B,N}^{-1/2}\right\|^2.
\]
\end{theorem}

\begin{proof}
Since \(D_NR_N=P_NM_N\),
\[
R_N^*H_NR_N
=
M_N^*P_ND_N(D_N^*D_N)^\dagger D_N^*P_NM_N
=
M_N^*P_NM_N.
\]
Also
\[
G_{\mathcal B,N}=M_N^*M_N
=
M_N^*P_NM_N+M_N^*(I-P_N)M_N.
\]
The last term is \(E_{\mathrm{coef},N}^*E_{\mathrm{coef},N}\).
\end{proof}
